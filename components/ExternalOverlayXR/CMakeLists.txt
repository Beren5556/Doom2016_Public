cmake_minimum_required(VERSION 3.20)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" OVERLAY_VERSION LIMIT_COUNT 1)
project(ExternalOverlayXRModule VERSION ${OVERLAY_VERSION} LANGUAGES CXX)
set(OPENXR_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/openxr/include" CACHE PATH "OpenXR SDK headers")
add_subdirectory(native)
add_executable(XrSmoke tests/XrSmoke.cpp)
target_compile_features(XrSmoke PRIVATE cxx_std_20)
target_compile_definitions(XrSmoke PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_include_directories(XrSmoke PRIVATE "${OPENXR_INCLUDE_DIR}" "${VULKAN_INCLUDE_DIR}")
target_compile_definitions(XrSmoke PRIVATE VK_NO_PROTOTYPES)
target_compile_options(XrSmoke PRIVATE /W4 /WX /EHsc /utf-8)
target_link_libraries(XrSmoke PRIVATE d3d11 d3d12 dxgi)
add_custom_command(TARGET XrSmoke POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:ExternalOverlayXR> $<TARGET_FILE_DIR:XrSmoke>)
enable_testing()
foreach(mode IN ITEMS d3d11 d3d12 index vive pico plus motion no-extensions)
  if(mode STREQUAL "d3d11")
    add_test(NAME xr-${mode} COMMAND XrSmoke)
  else()
    add_test(NAME xr-${mode} COMMAND XrSmoke --${mode})
  endif()
endforeach()
add_test(NAME xr-core11 COMMAND XrSmoke --api11 --no-extensions --pico)

add_test(NAME xr-vulkan COMMAND XrSmoke --vulkan)
add_test(NAME xr-vulkan2 COMMAND XrSmoke --vulkan2)
set_tests_properties(xr-vulkan xr-vulkan2 PROPERTIES TIMEOUT 30)
add_test(NAME xr-vulkan-abandoned-device COMMAND XrSmoke --vulkan)
set_tests_properties(xr-vulkan-abandoned-device PROPERTIES TIMEOUT 30 ENVIRONMENT "OVERLAY_TEST_ABANDON_DEVICE=1")
