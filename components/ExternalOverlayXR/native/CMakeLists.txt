cmake_minimum_required(VERSION 3.20)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../VERSION" OVERLAY_VERSION LIMIT_COUNT 1)
project(ExternalOverlayXR VERSION ${OVERLAY_VERSION} LANGUAGES CXX RC)
# Standalone build: no game SDK, UEVR, render plugin or AC7 source dependency.
set(OPENXR_INCLUDE_DIR "" CACHE PATH "Directory containing openxr/openxr.h")
if(NOT EXISTS "${OPENXR_INCLUDE_DIR}/openxr/openxr_loader_negotiation.h")
  message(FATAL_ERROR "Set OPENXR_INCLUDE_DIR to the OpenXR SDK include directory")
endif()
configure_file(Version.rc.in "${CMAKE_CURRENT_BINARY_DIR}/Version.rc" @ONLY)
add_library(ExternalOverlayXR SHARED OpenXRLayer.cpp "${CMAKE_CURRENT_BINARY_DIR}/Version.rc")
target_compile_features(ExternalOverlayXR PRIVATE cxx_std_20)
target_compile_definitions(ExternalOverlayXR PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
set(VULKAN_INCLUDE_DIR "" CACHE PATH "Directory containing vulkan/vulkan.h for this Vulkan pilot build")
if(NOT EXISTS "${VULKAN_INCLUDE_DIR}/vulkan/vulkan.h")
  message(FATAL_ERROR "Set VULKAN_INCLUDE_DIR to the Vulkan headers directory")
endif()
target_compile_definitions(ExternalOverlayXR PRIVATE VK_NO_PROTOTYPES)
target_include_directories(ExternalOverlayXR PRIVATE "${OPENXR_INCLUDE_DIR}" "${VULKAN_INCLUDE_DIR}")
target_compile_options(ExternalOverlayXR PRIVATE /W4 /WX /EHsc /utf-8)
target_link_libraries(ExternalOverlayXR PRIVATE d3d11 d3d12)
