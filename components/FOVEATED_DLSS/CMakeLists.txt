cmake_minimum_required(VERSION 3.24)
project(DlssFoveado VERSION 1.3.0 LANGUAGES CXX)

set(DLSS_FOVEADO_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(DLSS_FOVEADO_EXPECTED_VERSION "${PROJECT_VERSION}")
include(cmake/CheckEngineVersion.cmake)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/provenance.json"
  "${CMAKE_CURRENT_SOURCE_DIR}/include/dlss_foveado/Foveation.h")

add_library(dlss_foveado_core STATIC core/Geometry.cpp core/Config.cpp)
add_library(dlss_foveado::core ALIAS dlss_foveado_core)
set_target_properties(dlss_foveado_core PROPERTIES EXPORT_NAME core)
target_compile_features(dlss_foveado_core PUBLIC cxx_std_20)
target_include_directories(dlss_foveado_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
if(MSVC)
  target_compile_options(dlss_foveado_core PRIVATE /W4 /WX /EHsc /utf-8)
else()
  target_compile_options(dlss_foveado_core PRIVATE -Wall -Wextra -Werror)
endif()

option(DLSS_FOVEADO_BUILD_TESTS "Build independent contract tests" ${PROJECT_IS_TOP_LEVEL})
if(DLSS_FOVEADO_BUILD_TESTS)
  enable_testing()
  add_executable(dlss_foveado_contract_tests tests/Contracts.cpp)
  target_link_libraries(dlss_foveado_contract_tests PRIVATE dlss_foveado::core)
  add_test(NAME DlssFoveadoContracts COMMAND dlss_foveado_contract_tests)
  add_executable(dlss_foveado_four_side_tests tests/FourSides.cpp)
  target_compile_definitions(dlss_foveado_four_side_tests PRIVATE
    DLSS_FOVEADO_TEST_VERSION="${PROJECT_VERSION}")
  target_link_libraries(dlss_foveado_four_side_tests PRIVATE dlss_foveado::core)
  add_test(NAME DlssFoveadoFourSides COMMAND dlss_foveado_four_side_tests)
  add_executable(dlss_foveado_experimental_tests tests/Experimental.cpp)
  target_link_libraries(dlss_foveado_experimental_tests PRIVATE dlss_foveado::core)
  add_test(NAME DlssFoveadoExperimental COMMAND dlss_foveado_experimental_tests)
  add_executable(dlss_foveado_policy_tests tests/ConsumerPolicy.cpp)
  target_link_libraries(dlss_foveado_policy_tests PRIVATE dlss_foveado::core)
  add_test(NAME DlssFoveadoConsumerPolicy COMMAND dlss_foveado_policy_tests)
endif()

include(GNUInstallDirs)
option(DLSS_FOVEADO_BUILD_VULKAN "Build experimental Vulkan recording operations" OFF)
if(DLSS_FOVEADO_BUILD_VULKAN)
  add_subdirectory(backends/vulkan)
endif()
set(DLSS_FOVEADO_NGX_SDK "" CACHE PATH "Explicit NVIDIA NGX SDK root, optional")
option(DLSS_FOVEADO_BUILD_D3D11 "Build the Windows/D3D11 GPU backend" ${WIN32})
if(DLSS_FOVEADO_BUILD_D3D11)
  if(NOT WIN32)
    message(FATAL_ERROR "The D3D11 backend requires Windows")
  endif()
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/Operations.hlsl")
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/shaders/Operations.hlsl" DLSS_FOVEADO_SHADER_TEXT)
  configure_file(backends/d3d11/ShaderSource.h.in generated/ShaderSource.h @ONLY)
  add_library(dlss_foveado_d3d11 STATIC backends/d3d11/Operations.cpp backends/d3d11/Pipeline.cpp)
  add_library(dlss_foveado::d3d11 ALIAS dlss_foveado_d3d11)
  set_target_properties(dlss_foveado_d3d11 PROPERTIES EXPORT_NAME d3d11)
  target_link_libraries(dlss_foveado_d3d11 PUBLIC dlss_foveado_core d3d11 dxgi PRIVATE d3dcompiler)
  target_include_directories(dlss_foveado_d3d11 PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_compile_definitions(dlss_foveado_d3d11 PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
  if(MSVC)
    target_compile_options(dlss_foveado_d3d11 PRIVATE /W4 /WX /EHsc /utf-8)
  endif()
  install(TARGETS dlss_foveado_d3d11 EXPORT DlssFoveadoTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
  if(DLSS_FOVEADO_BUILD_TESTS)
    add_executable(dlss_foveado_gpu_tests tests/GpuOperations.cpp)
    target_link_libraries(dlss_foveado_gpu_tests PRIVATE dlss_foveado::d3d11)
    add_test(NAME DlssFoveadoGpuOperations COMMAND dlss_foveado_gpu_tests)
    add_executable(dlss_foveado_pipeline_tests tests/Pipeline.cpp)
    target_link_libraries(dlss_foveado_pipeline_tests PRIVATE dlss_foveado::d3d11)
    add_test(NAME DlssFoveadoPipeline COMMAND dlss_foveado_pipeline_tests)
    add_executable(dlss_foveado_dynamic_pipeline_tests tests/DynamicPipeline.cpp)
    target_link_libraries(dlss_foveado_dynamic_pipeline_tests PRIVATE dlss_foveado::d3d11)
    add_test(NAME DlssFoveadoDynamicPipeline COMMAND dlss_foveado_dynamic_pipeline_tests)
  endif()
endif()
set(DLSS_FOVEADO_WITH_NGX OFF)
if(DLSS_FOVEADO_NGX_SDK)
  set(DLSS_FOVEADO_WITH_NGX ON)
  if(NOT TARGET dlss_foveado_d3d11)
    message(FATAL_ERROR "NGX backend requires the D3D11 backend")
  endif()
  add_library(dlss_foveado_ngx STATIC backends/ngx/NgxBackend.cpp)
  add_library(dlss_foveado::ngx ALIAS dlss_foveado_ngx)
  set_target_properties(dlss_foveado_ngx PROPERTIES EXPORT_NAME ngx)
  target_compile_features(dlss_foveado_ngx PUBLIC cxx_std_20)
  add_library(DlssFoveado::NgxSdk UNKNOWN IMPORTED)
  set_target_properties(DlssFoveado::NgxSdk PROPERTIES IMPORTED_LOCATION "${DLSS_FOVEADO_NGX_SDK}/lib/Windows_x86_64/x64/nvsdk_ngx_d.lib")
  target_link_libraries(dlss_foveado_ngx PUBLIC dlss_foveado_d3d11 PRIVATE DlssFoveado::NgxSdk dxguid version psapi)
  target_include_directories(dlss_foveado_ngx PRIVATE "${DLSS_FOVEADO_NGX_SDK}/include")
  target_compile_options(dlss_foveado_ngx PRIVATE /W4 /EHsc /utf-8)
  install(TARGETS dlss_foveado_ngx EXPORT DlssFoveadoTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
  if(DLSS_FOVEADO_BUILD_TESTS)
    add_executable(dlss_foveado_ngx_hardware tests/NgxHardware.cpp)
    target_link_libraries(dlss_foveado_ngx_hardware PRIVATE dlss_foveado::ngx)
    # Explicit invocation: real NVIDIA GPU/runtime required. Never silently use WARP.
  endif()
endif()
install(TARGETS dlss_foveado_core EXPORT DlssFoveadoTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT DlssFoveadoTargets NAMESPACE dlss_foveado:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/DlssFoveado)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/DlssFoveadoConfig.cmake.in
  "${CMAKE_CURRENT_BINARY_DIR}/DlssFoveadoConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/DlssFoveado)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/DlssFoveadoConfigVersion.cmake"
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/DlssFoveadoConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/DlssFoveadoConfigVersion.cmake"
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/DlssFoveado)
