cmake_minimum_required(VERSION 3.24)
project(FoveationConsumer LANGUAGES CXX)
find_package(DlssFoveado 1.3 CONFIG REQUIRED)
add_executable(foveation_consumer main.cpp)
target_compile_definitions(foveation_consumer PRIVATE
  FOVEATION_EXPECTED_VERSION="${DlssFoveado_VERSION}")
target_link_libraries(foveation_consumer PRIVATE dlss_foveado::core)
if(TARGET dlss_foveado::d3d11)
  target_link_libraries(foveation_consumer PRIVATE dlss_foveado::d3d11)
  target_compile_definitions(foveation_consumer PRIVATE CONSUMER_D3D11)
endif()
if(TARGET dlss_foveado::ngx)
  target_link_libraries(foveation_consumer PRIVATE dlss_foveado::ngx)
  target_compile_definitions(foveation_consumer PRIVATE CONSUMER_NGX)
endif()
enable_testing()
add_test(NAME ExternalConsumer COMMAND foveation_consumer)
