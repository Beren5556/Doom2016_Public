cmake_minimum_required(VERSION 3.24)
project(ES2NeuralPost VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
add_library(ES2NeuralPost SHARED src/NeuralPost.cpp)
set_target_properties(ES2NeuralPost PROPERTIES OUTPUT_NAME DOOMNeuralPost)
target_include_directories(ES2NeuralPost PRIVATE include src)
target_link_libraries(ES2NeuralPost PRIVATE d3d11 d3d12 dxgi ole32)
add_executable(ES2NeuralWorker src/Worker.cpp)
set_target_properties(ES2NeuralWorker PROPERTIES OUTPUT_NAME DOOMNeuralWorker)
target_include_directories(ES2NeuralWorker PRIVATE include src vendor)
target_link_libraries(ES2NeuralWorker PRIVATE d3d11 d3d12 dxgi d3dcompiler advapi32)
add_library(NeuralForwarder SHARED vendor/nr-bridge/nr_forwarder.cpp)
set_target_properties(NeuralForwarder PROPERTIES OUTPUT_NAME "nvngx.dll_nr_bridge")
enable_testing()
add_executable(NeuralConfigTests tests/ConfigTests.cpp)
target_include_directories(NeuralConfigTests PRIVATE include)
add_test(NAME NeuralConfigTests COMMAND NeuralConfigTests)
add_executable(NeuralGpuTests tests/GpuTests.cpp)
target_include_directories(NeuralGpuTests PRIVATE include src)
target_link_libraries(NeuralGpuTests PRIVATE d3d11 d3d12 dxgi)
set(DOOM_ENGINE "${CMAKE_CURRENT_SOURCE_DIR}/../../upstream/KHARVOX")
add_executable(DoomNeuralVulkanTests tests/VulkanBridgeTests.cpp src/DoomNeuralVulkan.cpp)
target_compile_definitions(DoomNeuralVulkanTests PRIVATE VK_USE_PLATFORM_WIN32_KHR VK_NO_PROTOTYPES)
target_include_directories(DoomNeuralVulkanTests PRIVATE include src "${DOOM_ENGINE}/src/openxr" "${DOOM_ENGINE}/src/common" "${DOOM_ENGINE}/tests" "${DOOM_ENGINE}/third-party/vulkan-headers/include")
target_link_libraries(DoomNeuralVulkanTests PRIVATE d3d12 dxgi)
